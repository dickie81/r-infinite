#!/bin/bash
# lc.sh DIR MODULE: compile DIR/MODULE.lean to DIR/MODULE.olean (DIR first on LEAN_PATH, then the pilot build)
M=/tmp/claude-0/-home-user-r-infinite/db46f946-fd0e-58a3-baf6-61ea74b579d4/scratchpad/mathlib4
P=/home/user/r-infinite/tools/lean/riemann-pilot
export PATH="$HOME/.elan/bin:$PATH"
cd "$M" || exit 1
LP="$(lake env printenv LEAN_PATH)"
t0=$(date +%s)
out=$(LEAN_PATH="$LP:$1:$P/build" lean --root="$1" -o "$1/$2.olean" "$1/$2.lean" 2>&1)
rc=$?
echo "== $2 rc=$rc ($(( $(date +%s) - t0 ))s)"
[ -n "$out" ] && echo "$out" | head -${3:-200}
exit $rc
